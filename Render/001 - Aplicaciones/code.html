code SDK Quickstart

Inicializa el cliente de autenticación con PKCE habilitado en tu runtime web:

import { AuthShieldClient } from '@authshield/nextjs';

export const auth = new AuthShieldClient({
  domain: 'auth.acme.com',
  clientId: 'app_clnt_8f92a10b48e3',
  redirectUri: 'https://app.acme.com/callback',
  audience: 'https://api.acme.com/v1',
  scope: 'openid profile email offline_access',
  useRefreshTokens: true
});
